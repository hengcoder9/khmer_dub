អ្វីគ្រប់យ៉ាងសម្រាប់ការដាក់សំឡេង

ពីវីដេអូដើម ដល់វីដេអូខ្មែររួចរាល់ — ជំហាននីមួយៗធ្វើដោយស្វ័យប្រវត្តិ ហើយអ្នកអាចកែបានគ្រប់កន្លែង។

ស្តាប់ចេញអត្ថបទ

Whisper លើកាតក្រាហ្វិក ឬ Gemini — ចាប់សំដី និងពេលវេលាគ្រប់ឃ្លាដោយស្វ័យប្រវត្តិ។

បកប្រែជាខ្មែរ

Gemini, Claude ឬ OpenRouter — ឬបកប្រែក្រៅបណ្តាញ។ រក្សាឈ្មោះតួអង្គ និងរបៀបនិយាយ។

សំឡេងខ្មែរ និងចម្លងសំឡេង

សំឡេងខ្មែរធម្មជាតិ ឬចម្លងសំឡេងតួអង្គដើម។ សំឡេងតាមតួអង្គនីមួយៗ។

រក្សាតន្ត្រីដើម

បំបែកសំឡេងនិយាយចេញពីតន្ត្រី និងសំឡេងបរិយាកាស ដើម្បីឲ្យវីដេអូស្តាប់ទៅធម្មជាតិ។

កែវីដេអូ

កាត់ តម្រឹម លុប និងកាត់រូបភាព (crop) លើ timeline — ឬប្រើពាក្យបញ្ជា AI ដើម្បីកែ។

អក្សររត់ និងនាំចេញ

ដុតអក្សររត់ខ្មែរចូលវីដេអូ ឬនាំចេញ SRT ដែលត្រូវពេលវេលា។ MP4 រួចរាល់សម្រាប់ចែករំលែក។

ធ្វើបានក្នុង ៤ ជំហាន

ចុច Automate ដើម្បីឲ្យកម្មវិធីធ្វើទាំងអស់ ឬធ្វើម្តងមួយជំហាន។

បញ្ចូលវីដេអូ

បង្កើតគម្រោង ហើយបន្ថែមវគ្គ (episode) មួយ ឬច្រើន។

ស្តាប់ និងបកប្រែ

AI ចាប់សំដី ហើយបកប្រែជាខ្មែរ។ អ្នកកែឃ្លាណាក៏បាន។

បង្កើតសំឡេង

ជ្រើសសំឡេងសម្រាប់តួអង្គនីមួយៗ ហើយស្តាប់មុន។

នាំចេញ

ទទួលបាន MP4 ដែលមានសំឡេងខ្មែរ និងអក្សររត់ SRT។

តម្រូវការ

ដំណើរការលើកុំព្យូទ័ររបស់អ្នក — វីដេអូរបស់អ្នកមិនត្រូវបានផ្ញើទៅ server របស់យើងទេ។

កុំព្យូទ័រ

  • Windows 10 / 11 64-bit
  • RAM 8 GB ឡើងទៅ (ណែនាំ 16 GB)
  • កាត NVIDIA (ណែនាំ) — ធ្វើការលឿនជាងច្រើនដង
  • ទំហំទំនេរ 10 GB សម្រាប់ម៉ូដែល AI

គណនី AI (ជម្រើស)

  • Key Google Gemini ឥតគិតថ្លៃ — សម្រាប់បកប្រែ
  • ឬ Claude / OpenRouter
  • ឬប្រើ ក្រៅបណ្តាញ ទាំងស្រុង
  • Key ត្រូវបានរក្សាទុកដោយអ៊ិនគ្រីបនៅលើកុំព្យូទ័ររបស់អ្នក
